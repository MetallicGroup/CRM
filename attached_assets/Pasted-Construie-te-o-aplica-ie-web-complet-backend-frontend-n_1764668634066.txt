Formule obligatorii:

Valoare comision:

Valoare_Comision = Venit_TVA * (Comision/100)


Venit Brut:

Venit_Brut = Venit_TVA - Achizitie_TVA


TVA:

TVA = Venit_TVA * 0.19


Venit Net:

Venit_Net = Venit_Brut - TVA


Costuri Variabile TOTAL:

Costuri_Var = sum(Amortizare Auto → Diurne)


Profit Garduri:

Profit_Garduri = Venit_Gard - (Cost_Showroom + Cost_HQ + Cost_Productie + Cost_Indirecte)


Profit Acoperisuri:

Profit_Acoperisuri = Venit_Acoperis - (Cost_Showroom + Cost_HQ + Cost_Indirecte)


Profit General:

Profit_General = Venit_Net - Costuri_Var


Profit Final:

Profit_Final = Venit_Net - (Costuri_Var + Cost_Showroom + Cost_HQ + Cost_Productie + Cost_Indirecte + Costuri_Logistice)
